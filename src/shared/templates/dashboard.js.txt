  // Help icon: the text is shown by the single tooltip below (hover, focus or tap).
  function tip(key, cls) {
    return H[key] ? '<span class="h' + (cls ? " " + cls : "") + '" tabindex="0" role="button" data-h="' + esc(key) + '" aria-label="' + esc(H[key]) + '">?</span>' : "";
  }
  function num(v, d) {
    if (typeof v !== "number" || !isFinite(v)) return "–";
    var a = Math.abs(v);
    if (a >= 1e6) return (v / 1e6).toFixed(2) + " M";
    if (a >= 1e4) return (v / 1e3).toFixed(1) + " k";
    return d ? v.toFixed(d) : String(Math.round(v));
  }
  function pct(v) { return typeof v === "number" ? (v * 100).toFixed(1) + " %" : "–"; }
  function usd(v) { return typeof v === "number" ? "$" + v.toFixed(2) : "–"; }
  function dur(s) { return s < 120 ? Math.round(s) + " s" : s < 7200 ? Math.round(s / 60) + " min" : (s / 3600).toFixed(1) + " h"; }
  function hhmm(t) { return new Date(t).toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" }); }
  function lab(m, name, key, val) {
    for (var i = 0; i < m.labelled.length; i++) { var l = m.labelled[i]; if (l.name === name && l.labels[key] === val) return l.value; }
    return 0;
  }
  // Wait of a visitor arriving now, as shown in the tile and in the tab title.
  function wait(c) {
    var s = c.state, lag = c.metrics.gauges.wr_cursor_lag_seconds;
    if (s.mode === "off") return "off";
    return lag > 0 ? dur(lag) : "none";
  }
  function tiles(list) {
    return '<div class="tiles">' + list.map(function (t) {
      return '<div class="tile"><div class="k">' + esc(t[0]) + tip(t[0]) + '</div><div class="v">' + esc(t[1]) + "</div>" + (t[2] ? '<div class="s">' + esc(t[2]) + "</div>" : "") + "</div>";
    }).join("") + "</div>";
  }
  // lead: the help icon of each row comes before its label.
  function table(head, rows, lead) {
    var h = head ? "<tr>" + head.map(function (c, i) { return "<th" + (i ? ' class="n"' : "") + ">" + esc(c) + "</th>"; }).join("") + "</tr>" : "";
    return '<div class="scroll"><table>' + h + rows.map(function (r) {
      return "<tr>" + r.map(function (c, i) {
        if (i) return '<td class="n">' + esc(c) + "</td>";
        return "<td>" + (lead ? tip(c, "lead") + esc(c) : esc(c) + tip(c)) + "</td>";
      }).join("") + "</tr>";
    }).join("") + "</table></div>";
  }
  function box(title, inner, key) { return '<div class="box"><h2>' + esc(title) + tip(key || title) + "</h2>" + inner + "</div>"; }
  // Per-minute series: the last value is the current (partial) minute; only complete minutes
  // are plotted. Each chart keeps its values for the hover tooltip.
  var charts = {};
  function chart(hist, at, title, keys, labels) {
    var max = 1, n = 0, len = 0;
    var series = keys.map(function (k) { var a = hist[k] || []; len = a.length; return a.slice(0, -1); });
    series.forEach(function (s) { n = s.length; s.forEach(function (v) { if (v > max) max = v; }); });
    if (n < 2) return "";
    charts[title] = { labels: labels, series: series, n: n, t0: (Math.floor(at / 60000) - (len - 1)) * 60000 };
    var lines = series.map(function (s, i) {
      var pts = s.map(function (v, j) { return (j * 300 / (n - 1)).toFixed(1) + "," + (62 - v * 60 / max).toFixed(1); }).join(" ");
      return '<polyline class="l' + i + '" points="' + pts + '"/>';
    }).join("");
    var legend = series.map(function (s, i) {
      var sum = 0;
      s.forEach(function (v) { sum += v; });
      return '<span class="c' + i + '">' + esc(labels[i]) + ": " + num(s[n - 1] || 0) + "/min · 1 h " + num(sum) + "</span>";
    }).join("");
    return '<div class="chart" data-c="' + esc(title) + '"><div class="k">' + esc(title) + tip(title) + " <span class=\"mut\">(max " + num(max) + "/min)</span></div>" +
      '<svg viewBox="0 0 300 64" preserveAspectRatio="none" role="img" aria-label="' + esc(title) + '">' + lines +
      '<line x1="0" x2="0" y1="0" y2="64" visibility="hidden"/></svg><div class="lg">' + legend + "</div></div>";
  }
  function badges(c) {
    var s = c.state, b = [];
    if (s.mode !== "on") b.push('<span class="b ' + (s.mode === "off" ? "warn" : "bad") + '">mode ' + esc(s.mode) + tip("mode") + "</span>");
    if (s.paused) b.push('<span class="b bad">paused' + tip("paused") + "</span>");
    b.push(s.open ? '<span class="b ok">open' + tip("open") + "</span>" : '<span class="b warn">queueing' + tip("queueing") + "</span>");
    if (s.brakeFactor < 1) b.push('<span class="b warn">brake ×' + num(s.brakeFactor, 2) + tip("brake") + "</span>");
    return b.join("");
  }

  // Last answer of the coordinator: a failed refresh keeps it on screen (marked stale)
  // instead of replacing the whole page with an error.
  var last = null;
  function render(d) {
    $("room").textContent = d.room;
    var c = d.coordinator;
    if (c) last = { c: c, at: d.now };
    var shown = c || (last && last.c);
    $("meta").textContent = "v" + d.version + (shown ? " · config gen " + shown.state.gen : "") + " · " + new Date(d.now).toLocaleTimeString();
    if (c) updateControls(c.state);
    var stale = c ? "" : '<span class="b bad">coordinator did not answer' + (last ? ", data from " + new Date(last.at).toLocaleTimeString() : "") + tip("stale") + "</span>";
    $("badges").innerHTML = (shown ? badges(shown) : "") + stale;
    $("main").innerHTML = (shown ? coordinator(shown) : '<p class="bad">The coordinator did not answer.</p>') + (d.edge ? edge(d.edge) : "");
    // Wait first, to tell the rooms apart in a row of browser tabs.
    document.title = (shown ? (shown.state.paused ? "paused " : "") + wait(shown) + (c ? "" : " (stale)") + " · " : "") + d.room + " · Waiting room";
    // The refresh replaced the element under an open tooltip: follow its replacement.
    if (tipKey) {
      var a = document.querySelector('.h[data-h="' + tipKey.replace(/["\\]/g, "\\$&") + '"]');
      if (a) showTip(a); else hideTip();
    } else if (tipChart) {
      var sv = document.querySelector('.chart[data-c="' + tipChart.key.replace(/["\\]/g, "\\$&") + '"] svg');
      if (sv) chartTip(sv, tipChart.x); else hideTip();
    }
  }

  var tipKey = null, tipChart = null;
  function showAt(text, from, left, width, top, bottom) {
    var host = from.closest("dialog") || document.body;
    if (tipEl.parentNode !== host) host.appendChild(tipEl);
    tipEl.textContent = text;
    tipEl.hidden = false;
    var vw = document.documentElement.clientWidth;
    var x = Math.min(Math.max(8, left + width / 2 - tipEl.offsetWidth / 2), vw - tipEl.offsetWidth - 8);
    var y = bottom + 6;
    if (y + tipEl.offsetHeight > window.innerHeight - 8 && top - tipEl.offsetHeight - 6 > 8) y = top - tipEl.offsetHeight - 6;
    tipEl.style.left = x + "px";
    tipEl.style.top = y + "px";
  }
  function showTip(a) {
    var key = a.getAttribute("data-h");
    if (!H[key]) return;
    hideTip();
    tipKey = key;
    var r = a.getBoundingClientRect();
    showAt(H[key], a, r.left, r.width, r.top, r.bottom);
  }
  // Chart tooltip: the minute under the pointer, its time and the value of each series.
  function chartTip(svg, clientX) {
    var key = svg.parentNode.getAttribute("data-c"), ch = charts[key];
    if (!ch) return;
    var r = svg.getBoundingClientRect();
    var j = Math.min(ch.n - 1, Math.max(0, Math.round((clientX - r.left) / r.width * (ch.n - 1))));
    if (tipKey) hideTip();
    tipChart = { key: key, x: clientX };
    var line = svg.querySelector("line"), x = (j * 300 / (ch.n - 1)).toFixed(1);
    line.setAttribute("x1", x);
    line.setAttribute("x2", x);
    line.setAttribute("visibility", "visible");
    var text = hhmm(ch.t0 + j * 60000) + "\n" + ch.labels.map(function (l, i) { return l + ": " + num(ch.series[i][j]) + "/min"; }).join("\n");
    var px = r.left + j / (ch.n - 1) * r.width;
    showAt(text, svg, px, 0, r.top, r.bottom);
  }
  function hideTip() {
    tipKey = null;
    tipChart = null;
    tipEl.hidden = true;
    var l = document.querySelectorAll(".chart line");
    for (var i = 0; i < l.length; i++) l[i].setAttribute("visibility", "hidden");
  }
  function icon(e) { return e.target && e.target.closest ? e.target.closest(".h[data-h]") : null; }
  function plot(e) { return e.target && e.target.closest ? e.target.closest(".chart svg") : null; }
  document.addEventListener("mouseover", function (e) { var a = icon(e); if (a) showTip(a); });
  document.addEventListener("mouseout", function (e) { var a = icon(e); if (a && !a.contains(e.relatedTarget)) hideTip(); });
  document.addEventListener("pointermove", function (e) { var s = plot(e); if (s) chartTip(s, e.clientX); else if (tipChart) hideTip(); });
  document.addEventListener("pointerdown", function (e) { var s = plot(e); if (s) chartTip(s, e.clientX); });
  document.addEventListener("focusin", function (e) { var a = icon(e); if (a) showTip(a); });
  document.addEventListener("focusout", function (e) { if (icon(e)) hideTip(); });
  document.addEventListener("click", function (e) { var a = icon(e); if (a) showTip(a); else if (!plot(e)) hideTip(); });
  document.addEventListener("keydown", function (e) { if (e.key === "Escape") hideTip(); });
  window.addEventListener("scroll", function () { if (tipKey || tipChart) hideTip(); }, { passive: true });

    dialogs[q].addEventListener("close", function () { hideTip(); this.querySelector(".msg").textContent = ""; });
  }
  // Controls: static markup outside #main, so that the 10 s refresh never wipes what is being typed.
  function api(method, path, body) {
    return fetch("/__wr/dashboard/admin/" + path, {
      method: method,
      credentials: "same-origin",
      headers: { "Content-Type": "application/json" },
      body: body === undefined ? undefined : JSON.stringify(body)
    }).then(function (r) {
      return r.json().catch(function () { return {}; }).then(function (d) {
        if (!r.ok) throw new Error(d.error || "HTTP " + r.status);
        return d;
      });
    });
  }
  // Messages go to the open dialog, or under the controls.
  function say(text, ok) {
    var m = document.querySelector("dialog[open] .msg") || $("msg");
    m.textContent = text;
    m.className = "s msg " + (ok ? "msg-ok" : "msg-bad");
  }
  var busy = false;
  function act(label, method, path, body, done) {
    if (busy) return;
    busy = true;
    say(label + "…", true);
    api(method, path, body)
      .then(function (d) {
        say(label + ": done. Every data center applies it within 20 to 60 s.", true);
        if (done) done(d);
        load();
      })
      .catch(function (e) { say(label + ": " + e.message, false); })
      .then(function () { busy = false; });
  }
  var state = null;
  var dirty = {};
  function updateControls(s) {
    state = s;
    $("controls").hidden = false;
    var b = document.querySelectorAll("[data-mode]");
    for (var i = 0; i < b.length; i++) b[i].setAttribute("aria-pressed", String(b[i].getAttribute("data-mode") === s.mode));
    $("pause").textContent = s.paused ? "Resume admissions" : "Pause admissions";
    $("pause").setAttribute("aria-pressed", String(!!s.paused));
    var cfg = s.config || {};
    var f = $("f-rate");
    if (!dirty["f-rate"] && document.activeElement && f.contains(document.activeElement) === false) {
      f.admissionsPerMinute.value = cfg.admissionsPerMinute;
      f.sessionMinutes.value = cfg.sessionMinutes;
    }
    $("reff").textContent = "effective " + num(s.rEff, 1) + "/min";
    var wl = cfg.whitelist || [];
    $("wl-n").textContent = "(" + wl.length + ")";
    $("wl").innerHTML = wl.length === 0 ? '<p class="s">No entry.</p>' : '<div class="scroll"><table class="wl">' + wl.map(function (w) {
      return "<tr><td>" + esc(w.cidr) + "</td><td>" + esc(w.label) + '</td><td><button type="button" data-cidr="' + esc(w.cidr) + '">Remove</button></td></tr>';
    }).join("") + "</table></div>";
    var json = JSON.stringify(cfg, null, 2);
    if ($("cfg").textContent !== json) $("cfg").textContent = json;
  }
  var b = document.querySelectorAll("[data-mode]");
  for (var i = 0; i < b.length; i++) b[i].addEventListener("click", function () {
    var mode = this.getAttribute("data-mode");
    if (state && state.mode === mode) return;
    if (mode === "off" && !confirm("Turn the waiting room off? Every visitor goes straight to the site, without any control.")) return;
    if (mode === "queue_all" && !confirm("Send every new visitor through the queue, even when there is room?")) return;
    act("Mode " + mode, "PUT", "mode", { mode: mode });
  });
  $("pause").addEventListener("click", function () {
    var paused = !(state && state.paused);
    if (paused && !confirm("Pause admissions? Nobody new enters the site until you resume.")) return;
    act(paused ? "Pause" : "Resume", "PUT", "pause", { paused: paused });
  });
  ["f-rate", "f-cfg"].forEach(function (id) { $(id).addEventListener("input", function () { dirty[id] = true; }); });
  $("f-rate").addEventListener("submit", function (e) {
    e.preventDefault();
    var r = Number(this.admissionsPerMinute.value), d = Number(this.sessionMinutes.value), cfg = (state && state.config) || {};
    if (!confirm("Set the admission rate to " + r + "/min (now " + cfg.admissionsPerMinute + ") and the session duration to " + d + " min (now " + cfg.sessionMinutes + ")?")) return;
    act("Rate and session", "PATCH", "config", { admissionsPerMinute: r, sessionMinutes: d }, function () { dirty["f-rate"] = false; });
  });
  $("f-cfg").addEventListener("submit", function (e) {
    e.preventDefault();
    var form = this, patch;
    try { patch = JSON.parse(form.patch.value); } catch (x) { say("Settings: invalid JSON (" + x.message + ")", false); return; }
    if (typeof patch !== "object" || patch === null || Array.isArray(patch)) { say("Settings: a JSON object is expected", false); return; }
    if (!confirm("Apply " + JSON.stringify(patch) + "?")) return;
    act("Settings", "PATCH", "config", patch, function () { form.reset(); dirty["f-cfg"] = false; });
  });
  $("f-wl").addEventListener("submit", function (e) {
    e.preventDefault();
    var form = this;
    act("Whitelist " + form.cidr.value, "POST", "whitelist", { entries: [{ cidr: form.cidr.value.trim(), label: form.label.value.trim() }] }, function () { form.reset(); });
  });
  $("wl").addEventListener("click", function (e) {
    var t = e.target.closest ? e.target.closest("[data-cidr]") : null;
    if (!t) return;
    var cidr = t.getAttribute("data-cidr");
    if (!confirm("Remove " + cidr + " from the whitelist?")) return;
    act("Remove " + cidr, "DELETE", "whitelist", { cidrs: [cidr] });
  });
  $("f-pass").addEventListener("submit", function (e) {
    e.preventDefault();
    var form = this;
    act("Signed link", "POST", "pass", { host: form.host.value.trim().toLowerCase(), path: form.path.value, label: form.label.value.trim(), ttlSeconds: Number(form.ttl.value) }, function (d) {
      $("pass-url").value = d.url;
      $("pass-out").hidden = false;
      say("Signed link created, valid until " + new Date(d.expiresAt).toLocaleString() + ".", true);
    });
  });
  $("pass-copy").addEventListener("click", function () {
    var u = $("pass-url");
    u.select();
    if (navigator.clipboard) navigator.clipboard.writeText(u.value).then(function () { say("Link copied.", true); });
  });
  // Static help icons get their accessible text here (the dynamic ones carry it already).
  var hs = document.querySelectorAll(".h[data-h]");
  for (var j = 0; j < hs.length; j++) if (H[hs[j].getAttribute("data-h")]) hs[j].setAttribute("aria-label", H[hs[j].getAttribute("data-h")]);
  function load() {
    fetch("/__wr/dashboard/data", { cache: "no-store", credentials: "same-origin" })
      .then(function (r) {
        if (r.status === 401) { location.reload(); throw new Error("unauthorized"); }
        if (!r.ok) throw new Error("HTTP " + r.status);
        return r.json();
      })
      .then(render)
      .catch(function (e) { $("meta").textContent = "error: " + e.message; });
  }
  $("auto").addEventListener("click", function () { auto = !auto; this.textContent = "Auto-refresh: " + (auto ? "on" : "off"); if (auto) load(); });
  document.addEventListener("visibilitychange", function () { if (!document.hidden && auto) load(); });
  setInterval(function () { if (auto && !document.hidden) load(); }, REFRESH_MS);
  load();
})();
